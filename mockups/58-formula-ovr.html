Formula OVR / Attacco / Difesa

OVR = media degli OVR dei 5. Attacco e Difesa = media pesata per ruolo delle sotto-stat (come 2K: per un centro il sottocanestro pesa, il tiro da 3 quasi no). Tocca un giocatore per vedere i pesi accanto a ogni statistica.

L'algoritmo (pronto da innestare)
// Pesi per RUOLO: quanto conta ogni sotto-stat (1 = neutro).
// Es. Centro → sottocanestro pesa 3.0, tiro da 3 solo 0.3.
const PESI = {
  PG:{att:{tiro3:1.3, palleggio:1.6, visione:1.6, sottocanestro:0.5, ...}, def:{difPerimetro:1.6, rubate:1.5, ...}},
  PF:{att:{sottocanestro:2.6, tiro3:0.6, ...},                          def:{difInterna:1.5, rimbalzi:1.6, ...}},
  C :{att:{sottocanestro:3.0, tiro3:0.3, ...},                          def:{difInterna:2.0, stoppate:1.8, ...}},
  /* …SG, SF idem… */
};

// media PESATA delle sotto-stat di un gruppo, secondo il ruolo
function pesata(g, chiavi, pesiRuolo){
  let somma=0, pesi=0;
  for(const k of chiavi){ const w = pesiRuolo[k] ?? 1; somma += g.stat[k]*w; pesi += w; }
  return Math.round(somma / pesi);
}

// giocatore: ATT/DEF pesati per il suo ruolo · squadra: media dei 5
function valutaGiocatore(g){
  const p = PESI[g.ruolo];
  return { ovr:g.ovr, att:pesata(g,STAT_ATT,p.att), def:pesata(g,STAT_DEF,p.def) };
}
function valutaSquadra(rosa){
  const v = rosa.map(valutaGiocatore);
  return { ovr:media(v.map(x=>x.ovr)), att:media(v.map(x=>x.att)), def:media(v.map(x=>x.def)) };
}
